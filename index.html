/** * Contador de visitantes ÚNICOS con país — Zymar * * Uso: crea una Hoja de cálculo de Google nueva → Extensiones → Apps Script → * pega este código → Implementar → Nueva implementación → Tipo: Aplicación web * · Ejecutar como: Yo * · Quién tiene acceso: Cualquier persona * Copia la URL que termina en /exec y pégala en URL_CONTADOR dentro del index.html. * * Si cambias este código después, debes crear una NUEVA VERSIÓN en * Implementar → Administrar implementaciones → Editar → Nueva versión. */ var NOMBRE_HOJA = 'Visitas'; // Cambia esta frase por cualquier texto tuyo. Sirve para que los datos guardados no se puedan revertir. var SAL = 'zymar-cambia-esta-frase'; function doGet(e) { var p = (e && e.parameter) || {}; if (p.id) registrar_(p); return json_(resumen_()); } // Registra al visitante solo si NO se había visto antes (por navegador o por IP) function registrar_(p) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var hoja = hoja_(); var hid = hash_(String(p.id).substring(0, 100)); var hip = p.h ? hash_(String(p.h).substring(0, 60)) : ''; var porId = existe_(hoja, 1, hid); var porIp = hip ? existe_(hoja, 2, hip) : false; if (porId && (porIp || !hip)) return; // ya conocido: no se cuenta ni se guarda nada var cc = /^[A-Za-z]{2}$/.test(p.c || '') ? String(p.c).toUpperCase() : 'XX'; var cuenta = (porId || porIp) ? 0 : 1; // 1 = visitante nuevo, 0 = mismo visitante con otro dato (otra IP u otro navegador) hoja.appendRow([hid, hip, cc, cuenta, new Date()]); CacheService.getScriptCache().remove('resumen'); } finally { lock.releaseLock(); } } // Total de visitantes únicos y cuántos hay por país (los 8 primeros) function resumen_() { var cache = CacheService.getScriptCache(); var guardado = cache.get('resumen'); if (guardado) return JSON.parse(guardado); var hoja = hoja_(); var n = hoja.getLastRow() - 1; var total = 0, por = {}; if (n > 0) { hoja.getRange(2, 3, n, 2).getValues().forEach(function (f) { if (Number(f[1]) === 1) { total++; por[f[0]] = (por[f[0]] || 0) + 1; } }); } var paises = Object.keys(por) .filter(function (c) { return c !== 'XX'; }) .map(function (c) { return { c: c, n: por[c] }; }) .sort(function (a, b) { return b.n - a.n; }) .slice(0, 8); var res = { total: total, paises: paises }; cache.put('resumen', JSON.stringify(res), 300); return res; } function existe_(hoja, col, valor) { var filas = Math.max(hoja.getLastRow() - 1, 1); return hoja.getRange(2, col, filas, 1).createTextFinder(valor).matchEntireCell(true).findNext() !== null; } // Guarda solo huellas (hash), nunca la IP ni el identificador reales function hash_(texto) { var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, SAL + '|' + texto, Utilities.Charset.UTF_8); return 'h' + bytes.map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join(''); } function hoja_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = ss.getSheetByName(NOMBRE_HOJA); if (!hoja) { hoja = ss.insertSheet(NOMBRE_HOJA); hoja.appendRow(['Huella navegador', 'Huella IP', 'País', 'Cuenta', 'Fecha']); hoja.setFrozenRows(1); } return hoja; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
